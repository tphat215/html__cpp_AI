🔹 C++ là gì?

C++ là một ngôn ngữ lập trình bậc cao, được phát triển bởi Bjarne Stroustrup vào năm 1985 tại Bell Labs. Đây là phiên bản mở rộng của ngôn ngữ C với nhiều tính năng hiện đại như Lập trình hướng đối tượng (OOP), Lập trình tổng quát (Template) và Quản lý bộ nhớ thông minh.

🔹 Tại sao nên học C++?

🚀 Hiệu suất vượt trội

C++ là ngôn ngữ nhanh nhất trong số các ngôn ngữ bậc cao. Nó cho phép bạn kiểm soát bộ nhớ trực tiếp, tối ưu hóa code để chạy với tốc độ gần bằng Assembly. Điều này cực kỳ quan trọng trong các hệ thống đòi hỏi hiệu năng cao như game, AI, và phần mềm nhúng.

🎮 Vua của ngành Game

Hầu hết các engine game lớn đều được viết bằng C++: Unreal Engine, CryEngine, Unity (core), Source Engine. Nếu bạn muốn làm game AAA, C++ là kỹ năng bắt buộc. Tối ưu hóa đồ họa, xử lý vật lý, AI phức tạp đều dùng C++.

💰 Lương cao, nhu cầu lớn

Theo khảo sát, lương của lập trình viên C++ thuộc top đầu trong ngành IT. Các tập đoàn lớn như Google, Microsoft, Amazon, Facebook luôn săn đón kỹ sư C++ giỏi. Mức lương trung bình tại Việt Nam: 20-50 triệu/tháng cho senior, có thể lên tới 100 triệu/tháng cho các vị trí đặc biệt.

🤖 AI & Machine Learning

Các framework AI hàng đầu như TensorFlow, PyTorch, Caffe đều có core viết bằng C++. Lý do: C++ cung cấp hiệu năng tối đa cho việc xử lý hàng triệu phép tính toán học phức tạp. Nếu bạn muốn làm AI, C++ là công cụ đắc lực.

🖥️ Hệ điều hành & Nhúng

Windows, Linux, macOS kernel đều viết bằng C/C++. Các thiết bị IoT, vi điều khiển, hệ thống nhúng (embedded systems) dùng C++ vì nó tiết kiệm tài nguyên và có thể tương tác trực tiếp với phần cứng.

🧠 Nền tảng vững chắc

Học C++ giúp bạn hiểu sâu về cách máy tính hoạt động: bộ nhớ, con trỏ, quản lý tài nguyên. Khi đã giỏi C++, bạn sẽ dễ dàng học các ngôn ngữ khác như Java, Python, C#. Đây là nền tảng vững chắc cho mọi lập trình viên.

✨ Lớp và Đối tượng (OOP)

class Student {
private:
    // Thuộc tính private - chỉ truy cập trong class
    string name;
    int age;
    float gpa;

public:
    // Constructor - Khởi tạo đối tượng
    Student(string n, int a, float g) {
        name = n;
        age = a;
        gpa = g;
    }

    // Getter - Lấy giá trị
    string getName() { return name; }
    int getAge() { return age; }
    float getGPA() { return gpa; }

    // Setter - Gán giá trị
    void setName(string n) { name = n; }
    void setAge(int a) { age = a; }

    // Method - Hành động
    void display() {
        cout << "--- Thông tin sinh viên ---" << endl;
        cout << "Tên: " << name << endl;
        cout << "Tuổi: " << age << endl;
        cout << "GPA: " << gpa << endl;
        cout << "Xếp loại: ";

        if (gpa >= 3.6) cout << "Xuất sắc";
        else if (gpa >= 3.2) cout << "Giỏi";
        else if (gpa >= 2.5) cout << "Khá";
        else cout << "Trung bình";
        cout << endl;
    }

    // Destructor - Dọn dẹp (tự động gọi khi đối tượng bị hủy)
    ~Student() {
        cout << "Đã hủy sinh viên: " << name << endl;
    }
};

// Sử dụng class
Student s1("Nguyễn Văn A", 20, 3.8);

  • Object - Thể hiện cụ thể của class

✨ Kế thừa (Inheritance)

// Class cha (Base class)
class Animal {
protected:
    string name;
    int age;

public:
    Animal(string n, int a) : name(n), age(a) {}

    void eat() {
        cout << name << " đang ăn..." << endl;
    }

    void sleep() {
        cout << name << " đang ngủ..." << endl;
    }

    void display() {
        cout << "Tên: " << name << endl;
        cout << "Tuổi: " << age << endl;
    }
};

// Class con (Derived class) - Kế thừa từ Animal
class Dog : public Animal {
private:
    string breed;  // Giống chó

public:
    Dog(string n, int a, string b) : Animal(n, a), breed(b) {}

    // Phương thức riêng
    void bark() {
        cout << name << " sủa: Gâu gâu!" << endl;
    }

    // Ghi đè (override) phương thức
    void eat() {
        cout << name << " đang ăn xương..." << endl;
    }

    void display() {
        Animal::display();  // Gọi display của class cha
        cout << "Giống: " << breed << endl;
    }
};

// Class con khác
class Cat : public Animal {
public:
    Cat(string n, int a) : Animal(n, a) {}

    void meow() {
        cout << name << " kêu: Meo meo!" << endl;
    }
};

// Sử dụng
Dog myDog("Milo", 3, "Golden Retriever");
Cat myCat("Kitty", 2);

myDog.display();
myDog.eat();   // Gọi hàm đã ghi đè
myDog.bark();
myDog.sleep(); // Gọi hàm từ class cha

cout << endl;
myCat.display();
myCat.meow();
  • Kế thừa - Class con kế thừa từ class cha
  • Ghi đè - Viết lại phương thức của class cha
  • Đa hình - Cùng hành động nhưng khác cách thực hiện

📝 Ví dụ 2: Kiểm tra số nguyên tố

#include <iostream>
#include <cmath>
using namespace std;

bool isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) return false;
    }
    return true;
}

int main() {
    int n;
    cout << "Nhập số cần kiểm tra: ";
    cin >> n;
    
    if (isPrime(n)) {
        cout << n << " là số nguyên tố" << endl;
    } else {
        cout << n << " không là số nguyên tố" << endl;
    }
    
    // In ra các số nguyên tố từ 1 đến 100
    cout << "\nCác số nguyên tố từ 1 đến 100: ";
    for (int i = 2; i <= 100; i++) {
        if (isPrime(i)) {
            cout << i << " ";
        }
    }
    cout << endl;
    
    return 0;
}

📝 Ví dụ 3: Fibonacci với đệ quy và vòng lặp

#include <iostream>
using namespace std;

// Fibonacci với đệ quy
int fibonacci_recursive(int n) {
    if (n <= 1) return n;
    return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2);
}

// Fibonacci với vòng lặp (tối ưu hơn)
int fibonacci_loop(int n) {
    if (n <= 1) return n;
    
    int a = 0, b = 1, c;
    for (int i = 2; i <= n; i++) {
        c = a + b;
        a = b;
        b = c;
    }
    return b;
}

int main() {
    int n;
    cout << "Nhập số n: ";
    cin >> n;
    
    cout << "Fibonacci đệ quy (" << n << ") = " 
         << fibonacci_recursive(n) << endl;
    cout << "Fibonacci vòng lặp (" << n << ") = " 
         << fibonacci_loop(n) << endl;
    
    cout << "\nDãy Fibonacci từ 0 đến " << n << ": ";
    for (int i = 0; i <= n; i++) {
        cout << fibonacci_loop(i) << " ";
    }
    cout << endl;
    
    return 0;
}

        cout << "Nhập điểm Toán: "; cin >> math;
        cout << "Nhập điểm Văn: "; cin >> literature;
        cout << "Nhập điểm Anh: "; cin >> english;
    }

    double getAverage() {
        return (math + literature + english) / 3;
    }

    string getRank() {
        double avg = getAverage();
        if (avg >= 9.0) return "Xuất sắc";
        if (avg >= 8.0) return "Giỏi";
        if (avg >= 7.0) return "Khá";
        if (avg >= 5.0) return "Trung bình";
        return "Yếu";
    }

    void display() {
        cout << left << setw(10) << id
             << setw(20) << name
             << setw(8) << math
             << setw(8) << literature
             << setw(8) << english
             << setw(10) << fixed << setprecision(2) << getAverage()
             << setw(12) << getRank() << endl;
    }
};

int main() {
    vector<Student> students;
    int choice;
    
    do {
        cout << "\n=== QUẢN LÝ SINH VIÊN ===" << endl;
        cout << "1. Thêm sinh viên" << endl;
        cout << "2. Hiển thị danh sách" << endl;
        cout << "3. Thoát" << endl;
        cout << "Nhập lựa chọn: ";
        cin >> choice;
        
        if (choice == 1) {
            Student s;
            s.input();
            students.push_back(s);
            cout << "Đã thêm sinh viên thành công!" << endl;
        }
        else if (choice == 2) {
            if (students.empty()) {
                cout << "Danh sách trống!" << endl;
            } else {
                cout << "\n--- DANH SÁCH SINH VIÊN ---" << endl;
                cout << left << setw(10) << "Mã số"
                     << setw(20) << "Tên"
                     << setw(8) << "Toán"
                     << setw(8) << "Văn"
                     << setw(8) << "Anh"
                     << setw(10) << "TB"
                     << setw(12) << "Xếp loại" << endl;
                cout << "------------------------------------------------------------" << endl;
                
                for (Student &s : students) {
                    s.display();
                }
            }
        }
    } while (choice != 3);
    
    cout << "Đã thoát chương trình!" << endl;
    return 0;
}

🚀 Con trỏ (Pointer)

// Khai báo con trỏ
int number = 42;
int* ptr = &number;  // ptr lưu địa chỉ của number

cout << "Giá trị: " << number << endl;        // 42
cout << "Địa chỉ: " << &number << endl;      // 0x7ff... 
cout << "Giá trị qua con trỏ: " << *ptr << endl;  // 42

// Thay đổi giá trị qua con trỏ
*ptr = 100;
cout << "Giá trị mới: " << number << endl;  // 100

// Con trỏ NULL
int* nullPtr = nullptr;
if (nullPtr == nullptr) {
    cout << "Con trỏ rỗng" << endl;
}

// Cấp phát động
int* dynamicArray = new int[5];
for (int i = 0; i < 5; i++) {
    dynamicArray[i] = i * 10;
}

for (int i = 0; i < 5; i++) {
    cout << dynamicArray[i] << " ";
}
// Output: 0 10 20 30 40

// Giải phóng bộ nhớ
delete[] dynamicArray;
  • Con trỏ - Lưu địa chỉ bộ nhớ
  • *ptr - Lấy giá trị tại địa chỉ
  • &var - Lấy địa chỉ của biến
  • new/delete - Cấp phát/giải phóng động

🚀 Template - Lập trình tổng quát

// Template function
template <typename T>
T max(T a, T b) {
    return (a > b) ? a : b;
}

template <typename T>
void swap(T &a, T &b) {
    T temp = a;
    a = b;
    b = temp;
}

// Template class
template <typename T>
class Box {
private:
    T value;

public:
    Box(T v) : value(v) {}
    T getValue() { return value; }
    void setValue(T v) { value = v; }
    void display() {
        cout << "Giá trị: " << value << endl;
    }
};

// Sử dụng
cout << "max(10, 20) = " << max(10, 20) << endl;
cout << "max(3.14, 2.71) = " << max(3.14, 2.71) << endl;

int a = 5, b = 10;
swap(a, b);
cout << "a = " << a << ", b = " << b << endl;  // a = 10, b = 5

Box<int> intBox(100);
Box<string> strBox("Hello");

intBox.display();
strBox.display();

🚀 Lambda Expression (C++11)

#include <vector>
#include <algorithm>

vector<int> numbers = {5, 2, 8, 1, 9, 3};

// Lambda cơ bản
auto print = []() {
    cout << "Hello Lambda!" << endl;
};
print();

// Lambda có tham số
auto add = [](int a, int b) {
    return a + b;
};
cout << "Tổng: " << add(5, 3) << endl;

// Sắp xếp với lambda
sort(numbers.begin(), numbers.end(), [](int a, int b) {
    return a > b;  // Sắp xếp giảm dần
});

cout << "Sắp xếp giảm dần: ";
for (int x : numbers) {
    cout << x << " ";
}
// Output: 9 8 5 3 2 1

// Lambda với capture
int factor = 2;
auto multiply = [factor](int x) {
    return x * factor;
};
cout << "5 * 2 = " << multiply(5) << endl;

🚀 Exception Handling - Xử lý ngoại lệ

#include <stdexcept>

double divide(double a, double b) {
    if (b == 0) {
        throw runtime_error("Lỗi: Chia cho 0!");
    }
    return a / b;
}

int main() {
    double x, y;
    cout << "Nhập a và b: ";
    cin >> x >> y;
    
    try {
        double result = divide(x, y);
        cout << "Kết quả: " << result << endl;
    }
    catch (const runtime_error &e) {
        cout << "Bắt lỗi: " << e.what() << endl;
    }
    catch (...) {
        cout << "Lỗi không xác định!" << endl;
    }
    
    return 0;
}
  • try - Khối lệnh cần kiểm tra
  • catch - Bắt và xử lý ngoại lệ
  • throw - Ném ngoại lệ
