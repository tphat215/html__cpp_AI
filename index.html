📝 Ví dụ 2: Kiểm tra số nguyên tố

C++ 
#include <iostream>
#include <cmath>
using namespace std;

bool isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) return false;
    }
    return true;
}

int main() {
    int n;
    cout << "Nhập số cần kiểm tra: ";
    cin >> n;

    if (isPrime(n)) {
        cout << n << " là số nguyên tố";
    } else {
        cout << n << " không là số nguyên tố";
    }

    return 0;
}

📝 Ví dụ 3: Lớp Học Sinh

C++ 
#include <iostream>
using namespace std;

class Student {
private:
    string name;
    float score;

public:
    // Constructor
    Student(string n, float s) {
        name = n;
        score = s;
    }

    // Getter
    string getName() { return name; }
    float getScore() { return score; }

    // Method
    void display() {
        cout << "Sinh viên: " << name << endl;
        cout << "Điểm: " << score << endl;
        cout << "Xếp loại: ";

        if (score >= 8.5) cout << "Giỏi";
        else if (score >= 7.0) cout << "Khá";
        else if (score >= 5.0) cout << "Trung bình";
        else cout << "Yếu";
        cout << endl;
    }
};

int main() {
    Student s1("Nguyễn Văn A", 8.7);
    Student s2("Trần Thị B", 6.5);

    s1.display();
    cout << endl;
    s2.display();

    return 0;
}
